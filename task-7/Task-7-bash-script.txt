#!/bin/bash

MY_DIR="task7-logs"
DATE=$(date +%F)
LOG_FILE="$MY_DIR/my-app-logs_$DATE.log"
mkdir "$MY_DIR"


# ===== Rotate old logs (older than 7 days) =====
find "$MY_DIR" -type f -name "*.log" -mtime +7 -delete

exec >> "$LOG_FILE" 2>&1

echo "HELLO"


echo "1.host name: "$(hostname)
echo "2.date and time: "$(date)
echo "3.uptime: "$(uptime)
echo "4.logged in users: " $(who)

echo "5.cpu usage: "
top -bn1 | grep "Cpu(s)" | awk '{print "CPU Idle:", $8 " - CPU Used:", 100-$8}'

echo "6.memory usage: "
free | awk 'NR==2{print "Memory used:", $3, "Total Mem:",$2, "%Mem used:"$3/$2*100}'

echo "7.disk usage: "
df -h | awk 'NR>1{gsub("%",""); if($5>10) print " Alert "  $6  " is "  $5  }'

echo "8.top 5 processes consuming the most CPU: "
ps -eo pid,comm,%cpu --sort=-%cpu | head -6

echo "9.process is consuming more than 70% CPU: "
ps -eo pid,comm,%cpu --sort=-%cpu | awk -v d="$(date)" '$3 > 70 {print d, $0}'
~
~
